function(haf_add_example name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE haf_core haf_adapters)
  if(TARGET haf_adapter_cuda)
    target_link_libraries(${name} PRIVATE haf_adapter_cuda)
    target_compile_definitions(${name} PRIVATE HAF_HAVE_CUDA_ADAPTER=1)
  endif()
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  haf_apply_warnings(${name})
endfunction()
